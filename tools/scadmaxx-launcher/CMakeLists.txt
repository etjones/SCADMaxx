cmake_minimum_required(VERSION 3.24)
find_program(LAUNCHER_POWERSHELL NAMES pwsh powershell REQUIRED)
execute_process(
  COMMAND "${LAUNCHER_POWERSHELL}" -NoProfile -NonInteractive -File "${CMAKE_CURRENT_SOURCE_DIR}/Get-Version.ps1"
  RESULT_VARIABLE version_result OUTPUT_VARIABLE LAUNCHER_VERSION
  ERROR_VARIABLE version_error OUTPUT_STRIP_TRAILING_WHITESPACE)
if(NOT version_result EQUAL 0)
  message(FATAL_ERROR "Cannot read launcher version.xml: ${version_error}")
endif()
string(REGEX MATCH "^[0-9]+\\.[0-9]+\\.[0-9]+" LAUNCHER_NUMERIC_VERSION "${LAUNCHER_VERSION}")
project(SCADMaxxLauncher VERSION ${LAUNCHER_NUMERIC_VERSION} LANGUAGES CXX)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS version.xml Get-Version.ps1)
configure_file(launcher-version.h.in launcher-version.h @ONLY)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
if(MSVC)
  add_compile_options(/utf-8)
endif()
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Network Concurrent Test)
add_library(launcher-core core.cpp)
target_link_libraries(launcher-core PUBLIC Qt6::Core)
target_include_directories(launcher-core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
if(WIN32)
  target_link_libraries(launcher-core PUBLIC version shlwapi shell32)
endif()
qt_add_executable(scadmaxx-launcher WIN32 main.cpp)
qt_add_resources(scadmaxx-launcher launcher-assets PREFIX "/" FILES launcher.ico)
if(WIN32)
  enable_language(RC)
  configure_file(launcher-version.rc.in launcher-version.rc @ONLY)
  target_sources(scadmaxx-launcher PRIVATE launcher.rc "${CMAKE_CURRENT_BINARY_DIR}/launcher-version.rc")
endif()
target_include_directories(scadmaxx-launcher PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(scadmaxx-launcher PRIVATE launcher-core Qt6::Widgets Qt6::Network Qt6::Concurrent)
include(CTest)
if(BUILD_TESTING)
  add_test(NAME launcher-version COMMAND "${LAUNCHER_POWERSHELL}" -NoProfile -NonInteractive -File "${CMAKE_CURRENT_SOURCE_DIR}/Test-Version.ps1")
  qt_add_executable(launcher-tests tests.cpp)
  set_target_properties(launcher-tests PROPERTIES WIN32_EXECUTABLE FALSE)
  target_link_libraries(launcher-tests PRIVATE launcher-core Qt6::Test Qt6::Widgets Qt6::Network)
  add_test(NAME launcher-core COMMAND launcher-tests)
  add_test(NAME launcher-ui COMMAND scadmaxx-launcher --smoke-test)
  set_tests_properties(launcher-ui PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  if(WIN32 AND VCPKG_INSTALLED_DIR)
    # Build-tree tests need the shared Qt plugins; deployment supplies its own qt.conf.
    set_property(TEST launcher-ui APPEND PROPERTY ENVIRONMENT
      "QT_QPA_PLATFORM_PLUGIN_PATH=${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/Qt6/plugins/platforms"
      "QT_PLUGIN_PATH=${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/Qt6/plugins")
  endif()
endif()
install(TARGETS scadmaxx-launcher RUNTIME DESTINATION .)
if(WIN32 AND VCPKG_INSTALLED_DIR)
  # This workspace's vcpkg Qt installation intentionally has no windeployqt.
  find_program(LAUNCHER_DUMPBIN dumpbin REQUIRED)
  set(LAUNCHER_DEPENDENCIES "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}")
  configure_file(Deploy.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/Deploy.cmake" @ONLY)
  install(SCRIPT "${CMAKE_CURRENT_BINARY_DIR}/Deploy.cmake")
  set(CMAKE_INSTALL_SYSTEM_RUNTIME_DESTINATION .)
  include(InstallRequiredSystemLibraries)
else()
  qt_generate_deploy_app_script(TARGET scadmaxx-launcher OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
  install(SCRIPT ${deploy_script})
endif()
install(FILES README.md USAGE.md version.xml DESTINATION .)
